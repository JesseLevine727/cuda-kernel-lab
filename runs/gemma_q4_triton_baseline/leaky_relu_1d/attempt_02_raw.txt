import torch
import torch.nn as nn
import triton
import triton.language as tl

@triton.jit
def leaky_relu_kernel(x, out, negative_slope: tl.constexpr, n: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < n
    
    x_val = tl.load(x + offs, mask=mask, other=0.0)
    
    # Leaky ReLU logic: if x < 0, return negative_slope * x, else x
    res = tl.where(x_val > 0.0, x_val, x_val * negative_slope)
    
    tl.store(out + offs, res, mask=mask)

class ModelNew(nn.Module):
    def __init__(self):
        super().__init__()

    def forward(self, x: torch.Tensor, negative_slope: float) -> torch.Tensor:
        x = x.contiguous()
        out = torch.empty_like(x)
        n = x.numel()
        
        # Use a power of 2 for the block size
        BLOCK_SIZE = 1024
        
        grid = lambda meta: (triton.cdiv(n, meta["BLOCK"]),)
        
        leaky_relu_kernel[grid](
            x, 
            out, 
            negative_slope, 
            n, 
            BLOCK=BLOCK_SIZE
        )
        return out
